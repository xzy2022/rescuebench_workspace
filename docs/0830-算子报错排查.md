## 第三次 完全禁用 cudnn
```bash
PROJECT_ROOT=/root/autodl-tmp/rescuebench_workspace
REPO="$PROJECT_ROOT/repos/RescueBench"
ENV_PATH="$PROJECT_ROOT/conda/envs/rescue-nomad"
RUN_ROOT="$PROJECT_ROOT/runs/nomad-debug-no-cudnn-0830-p1"
LOG="$PROJECT_ROOT/runs/nomad-debug-no-cudnn-0830-p1.log"

cd "$REPO"

"$PROJECT_ROOT/bin/rescue-run" \
    env CUDA_LAUNCH_BLOCKING=1 PYTHONUNBUFFERED=1 \
    "$ENV_PATH/bin/python" -u -c \
    'import runpy, torch; torch.backends.cudnn.enabled=False; print("CUDNN_ENABLED=", torch.backends.cudnn.enabled, flush=True); runpy.run_path("benchmark/rescue_benchmark.py", run_name="__main__")' \
    --model nomad \
    --levels 0 \
    --episodes 1 \
    --point-ids 1 \
    --no-collision \
    --enable-trajectory \
    --output "$RUN_ROOT" \
    2>&1 | tee "$LOG"
    
# 解决报错：
# `runpy.run_path()` 不会自动把 `benchmark/` 加入 Python 模块搜索路径，因此找不到 `agents`
RUN_ROOT="$PROJECT_ROOT/runs/nomad-debug-no-cudnn-0830-p1-v2"
LOG="$PROJECT_ROOT/runs/nomad-debug-no-cudnn-0830-p1-v2.log"

cd "$REPO"

"$PROJECT_ROOT/bin/rescue-run" \
    env CUDA_LAUNCH_BLOCKING=1 PYTHONUNBUFFERED=1 \
    "$ENV_PATH/bin/python" -u -c \
    'import os, runpy, sys, torch; sys.path.insert(0, os.path.abspath("benchmark")); torch.backends.cudnn.enabled=False; print("CUDNN_ENABLED=", torch.backends.cudnn.enabled, flush=True); runpy.run_path("benchmark/rescue_benchmark.py", run_name="__main__")' \
    --model nomad \
    --levels 0 \
    --episodes 1 \
    --point-ids 1 \
    --no-collision \
    --enable-trajectory \
    --output "$RUN_ROOT" \
    2>&1 | tee "$LOG"
```
